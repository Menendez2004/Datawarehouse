// Ejecutar en SSMS: Connect -> Analysis Services -> New Query -> MDX
// Los nombres de dimensiones/atributos corresponden a los definidos en
// docs/guia_cubo_ssas.md. Ajuste si usó nombres distintos.
// Los resultados deben coincidir con sql/03_problemas_negocio.sql.
// =====================================================================

// Problema 1: Ventas y utilidad por año y trimestre
SELECT
    { [Measures].[Ventas], [Measures].[Utilidad], [Measures].[Margen Pct] } ON COLUMNS,
    NON EMPTY [Fecha].[Calendario].[Trimestre].MEMBERS ON ROWS
FROM [Ventas Internet];

// Problema 2: Categorías y subcategorías
SELECT
    { [Measures].[Unidades], [Measures].[Ventas], [Measures].[Utilidad], [Measures].[Margen Pct] } ON COLUMNS,
    NON EMPTY ORDER( [Producto].[Categorias].[Subcategoria].MEMBERS,
                     [Measures].[Ventas], BDESC ) ON ROWS
FROM [Ventas Internet];

// Problema 3: Top 10 productos por utilidad
SELECT
    { [Measures].[Ventas], [Measures].[Utilidad] } ON COLUMNS,
    TOPCOUNT( [Producto].[Producto].[Producto].MEMBERS, 10, [Measures].[Utilidad] ) ON ROWS
FROM [Ventas Internet];

// Problema 4: Ventas por territorio
SELECT
    { [Measures].[Ventas], [Measures].[Ordenes] } ON COLUMNS,
    NON EMPTY [Territorio].[Territorios].[Region].MEMBERS ON ROWS
FROM [Ventas Internet];

// Problema 5: Perfil del cliente
SELECT
    { [Measures].[Ventas] } ON COLUMNS,
    NON EMPTY [Cliente].[Genero].[Genero].MEMBERS
            * [Cliente].[Estado Civil].[Estado Civil].MEMBERS
            * [Cliente].[Educacion].[Educacion].MEMBERS ON ROWS
FROM [Ventas Internet];

// Problema 6: Promociones
SELECT
    { [Measures].[Ventas], [Measures].[Unidades], [Measures].[Utilidad] } ON COLUMNS,
    NON EMPTY [Promocion].[Promocion].[Promocion].MEMBERS ON ROWS
FROM [Ventas Internet];
